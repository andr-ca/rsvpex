# Phase 11: Testing & CI — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** The GitHub Actions CI pipeline enforces lint, typecheck, ≥80% unit coverage (100% on critical modules), and axe-core accessibility stubs — all gates pass before merge. Every exported function carries `@req` JSDoc tags enforced by a custom ESLint rule.

**Architecture:** ESLint flat config with TypeScript-aware rules + a custom rule that enforces `@req` JSDoc tags on all exported declarations in `domain/`, `routes/`, `handlers/`, and `middleware/`. Prettier handles formatting separately. GitHub Actions CI runs four sequential jobs: lint → typecheck → unit test (with coverage thresholds) → placeholder E2E job. Playwright + axe-core configs are created as stubs for future E2E work (full E2E requires a running wrangler dev server, which is non-trivial in CI).

**Tech Stack:** ESLint 9 (flat config), Prettier 3, `@typescript-eslint` 8, Vitest 4 + `@cloudflare/vitest-pool-workers`, GitHub Actions, Playwright (stub config)

### Task 2: Custom ESLint Rule — `@req` Tag Enforcement

**Files:**
- Create: `app/eslint-rules/require-req-tag.js`
- Create: `app/eslint-rules/require-req-tag.test.js`

- [ ] **Step 1: Write the custom ESLint rule**

The rule checks every `ExportNamedDeclaration` and `ExportDefaultDeclaration` in files matching `src/(domain|routes|handlers|middleware)/`. It verifies that the export (or its declaration) has a leading JSDoc comment containing `@req`. Type exports (`export type`, `export interface`) are exempt. Constant exports that are not functions (e.g. `export const IMPORT_HEADERS = [...]`) are exempt.

```javascript
// app/eslint-rules/require-req-tag.js

/** @type {import('eslint').Rule.RuleModule} */
const rule = {
  meta: {
    type: 'suggestion',
    docs: {
      description: 'Require @req JSDoc tag on exported functions in domain/, routes/, handlers/, middleware/',
      category: 'Documentation',
    },
    messages: {
      missingReqTag:
        'Exported function "{{name}}" is missing a @req JSDoc tag referencing a requirement ID (e.g. @req CAP-01).',
    },
    schema: [],
  },
  create(context) {
    const filename = context.filename || context.getFilename()

    // Only enforce in target directories
    const targetDirs = /src\/(domain|routes|handlers|middleware)\//
    if (!targetDirs.test(filename)) {
      return {}
    }

    /**
     * Check if a node has a leading JSDoc comment with @req tag.
     * Walks up through file-level comments too.
     */
    function hasReqTag(node) {
      const sourceCode = context.sourceCode || context.getSourceCode()
      const comments = sourceCode.getCommentsBefore(node)
      for (const comment of comments) {
        if (comment.type === 'Block' && /@req\s+\S/.test(comment.value)) {
          return true
        }
      }
      return false
    }

    /**
     * Check if a node is a file-level JSDoc (appears at the very top of the file,
     * before any other statements).
     */
    function hasFileLeveReqTag() {
      const sourceCode = context.sourceCode || context.getSourceCode()
      const program = sourceCode.ast
      if (program.body.length === 0) return false
      const firstNode = program.body[0]
      const comments = sourceCode.getCommentsBefore(firstNode)
      for (const comment of comments) {
        if (comment.type === 'Block' && /@req\s+\S/.test(comment.value)) {
          return true
        }
      }
      return false
    }

    /**
     * Determine if a declaration is a function-like export.
     * We skip pure type exports and non-function constants.
     */
    function isFunctionLikeExport(node) {
      if (!node) return false
      // Function declaration
      if (node.type === 'FunctionDeclaration') return true
      // Variable declaration with arrow function or function expression
      if (node.type === 'VariableDeclaration') {
        return node.declarations.some(
          (d) =>
            d.init &&
            (d.init.type === 'ArrowFunctionExpression' ||
              d.init.type === 'FunctionExpression'),
        )
      }
      // TSTypeAliasDeclaration, TSInterfaceDeclaration — skip
      if (
        node.type === 'TSTypeAliasDeclaration' ||
        node.type === 'TSInterfaceDeclaration'
      ) {
        return false
      }
      return false
    }

    function getExportName(node) {
      if (node.declaration) {
        if (node.declaration.id) return node.declaration.id.name
        if (node.declaration.declarations && node.declaration.declarations[0]) {
          return node.declaration.declarations[0].id.name
        }
      }
      return '<anonymous>'
    }

    return {
      ExportNamedDeclaration(node) {
        // Skip type-only exports
        if (node.exportKind === 'type') return
        // Skip re-exports (export { x } from 'y')
        if (node.source) return
        // Skip non-function exports
        if (node.declaration && !isFunctionLikeExport(node.declaration)) return

        if (!hasReqTag(node) && !hasFileLeveReqTag()) {
          context.report({
            node,
            messageId: 'missingReqTag',
            data: { name: getExportName(node) },
          })
        }
      },
      ExportDefaultDeclaration(node) {
        // Default exports (routers) — check for @req tag
        if (!hasReqTag(node) && !hasFileLeveReqTag()) {
          context.report({
            node,
            messageId: 'missingReqTag',
            data: { name: 'default' },
          })
        }
      },
    }
  },
}

module.exports = rule
```

// app/eslint-rules/require-req-tag.test.js
const { RuleTester } = require('eslint')
const rule = require('./require-req-tag')

const ruleTester = new RuleTester({
  languageOptions: {
    ecmaVersion: 2022,
    sourceType: 'module',
  },

ruleTester.run('require-req-tag', rule, {
  valid: [
    // File-level @req tag covers all exports

      filename: 'src/domain/capacity.ts',

    // Function-level @req tag

    {
      code: '/** @req ADMIN-01 */\nexport const requireAdmin = () => {}',
      filename: 'src/middleware/requireAdmin.ts',

    // Default export with file-level @req
    {
