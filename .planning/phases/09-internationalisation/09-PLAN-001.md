# Phase 9: Internationalisation — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Public RSVP forms are displayed in the event's configured language (English, French, or Spanish), falling back to the browser's language when the event locale is not set.

**Architecture:** A pure `i18n` module provides a flat translation dictionary (`en`, `fr`, `es`) and a `t(key, locale)` lookup function. A pure `resolveLocale(eventLocale, acceptLanguageHeader)` function resolves the effective locale. All public-facing HTML renderers (`rsvpForm.ts`, `rsvpThankYou.ts`, `rsvpSubmit.ts` capacity-full page, `rsvpPatch.ts`) thread the resolved locale and call `t()` instead of hardcoded English strings. Admin routes remain English-only.

**Tech Stack:** TypeScript, Hono, Vitest + `@cloudflare/vitest-pool-workers`

**Requirements:** I18N-01

- [ ] **Step 2: Create `resolve.ts` with locale resolution logic**

```typescript
import { SUPPORTED_LOCALES, type SupportedLocale } from './translations'

/**
 * Resolve the effective locale for a public page.
 *
 * Priority:
 * 1. Event's configured locale (if supported)
 * 2. Browser Accept-Language header (first supported match)
 * 3. English fallback
 *
 * @req I18N-01 — event locale with Accept-Language fallback
 */
export function resolveLocale(
  eventLocale: string | null | undefined,
  acceptLanguage: string | null | undefined,
): SupportedLocale {
  // 1. Event locale takes priority
  if (eventLocale && SUPPORTED_LOCALES.includes(eventLocale as SupportedLocale)) {
    return eventLocale as SupportedLocale
  }

  // 2. Parse Accept-Language header
  if (acceptLanguage) {
    const parsed = acceptLanguage
      .split(',')
      .map((part) => {
        const [lang, q] = part.trim().split(';q=')
        return { lang: lang.trim().split('-')[0].toLowerCase(), q: q ? parseFloat(q) : 1.0 }
      })
      .sort((a, b) => b.q - a.q)

    for (const { lang } of parsed) {
      if (SUPPORTED_LOCALES.includes(lang as SupportedLocale)) {
        return lang as SupportedLocale
      }
    }
  }

  // 3. Default to English
  return 'en'
}
```

- [ ] **Step 3: Create `index.ts` barrel export**

```typescript
import { translations, type SupportedLocale, SUPPORTED_LOCALES } from './translations'
export { resolveLocale } from './resolve'
export type { SupportedLocale }
export { SUPPORTED_LOCALES }

/**
 * Translate a key for the given locale.
 * Falls back to English, then returns the key itself if not found.
 *
 * @req I18N-01
 */
export function t(key: string, locale: SupportedLocale): string {
  return translations[locale]?.[key] ?? translations.en[key] ?? key
}
```

    await env.DB.exec('DELETE FROM rsvps')
    await env.DB.exec('DELETE FROM events')
  })

  async function seedEvent(overrides: Record<string, unknown> = {}) {
    const defaults = {
      id: 'evt-i18n',
      slug: 'test-i18n',
      title: 'Fête Internationale',
      start_at: '2026-06-15T14:00:00Z',
      status: 'published',
      locale: 'en',
      visibility: 'public',
      is_kids_event: 0,
      allow_status_choice: 1,
      max_party_size_per_rsvp: 10,
      enable_waitlist: 0,
      enable_heuristic_dup_check: 0,
      notify_via_email: 0,
      notify_via_sms: 0,
      questions: '[]',
    }
    const row = { ...defaults, ...overrides }
    const cols = Object.keys(row)
    const placeholders = cols.map(() => '?').join(', ')
    await env.DB.prepare(`INSERT INTO events (${cols.join(', ')}) VALUES (${placeholders})`)
      .bind(...Object.values(row))
      .run()
  }

  it('renders French form when event locale is fr', async () => {
    await seedEvent({ locale: 'fr' })
    const res = await app.fetch(new Request('http://localhost/rsvp/test-i18n'), env)
    const html = await res.text()
    expect(html).toContain('Nom complet')
    expect(html).toContain('Envoyer mon RSVP')
    expect(html).toContain('lang="fr"')
  })

  it('renders Spanish form when event locale is es', async () => {
    await seedEvent({ locale: 'es', slug: 'test-es', id: 'evt-es' })
    const res = await app.fetch(new Request('http://localhost/rsvp/test-es'), env)
    const html = await res.text()
    expect(html).toContain('Nombre completo')
    expect(html).toContain('Enviar mi RSVP')
    expect(html).toContain('lang="es"')
  })

  it('falls back to Accept-Language when event locale is empty', async () => {
    await seedEvent({ locale: '' })
    const res = await app.fetch(
      new Request('http://localhost/rsvp/test-i18n', {
        headers: { 'Accept-Language': 'fr-FR,fr;q=0.9' },
      }),
      env,
    )
    const html = await res.text()
    expect(html).toContain('Nom complet')
    expect(html).toContain('lang="fr"')
  })

  it('renders French thank-you page for French event', async () => {
    await seedEvent({ locale: 'fr' })
    // Seed an RSVP
    await env.DB.prepare(
      `INSERT INTO rsvps (id, event_id, name, email, adults, parents_count, siblings_count, children_count, dietary, answers, status, rsvp_token)
       VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)`,
    ).bind('rsvp-fr', 'evt-i18n', 'Marie', 'marie@test.com', 1, 0, 0, 0, '[]', '{}', 'attending', 'tok-fr').run()

    const res = await app.fetch(new Request('http://localhost/rsvp/thank-you?rid=tok-fr'), env)
    const html = await res.text()
    expect(html).toContain('RSVP confirmé')
    expect(html).toContain('Votre RSVP')
    expect(html).toContain('lang="fr"')
  })

  it('renders French capacity-full page', async () => {
    await seedEvent({ locale: 'fr', max_guests_total: 1 })
    // Fill capacity
    await env.DB.prepare(
      `INSERT INTO rsvps (id, event_id, name, email, adults, parents_count, siblings_count, children_count, dietary, answers, status, rsvp_token)
       VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)`,
    ).bind('rsvp-fill', 'evt-i18n', 'Jean', 'jean@test.com', 1, 0, 0, 0, '[]', '{}', 'attending', 'tok-fill').run()

    const form = new URLSearchParams({
      name: 'Pierre',
      email: 'pierre@test.com',
      adults: '1',
    })
    const res = await app.fetch(
      new Request('http://localhost/rsvp/test-i18n', {
        method: 'POST',
        headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
        body: form.toString(),
      }),
      env,
    )
    expect(res.status).toBe(409)
    const html = await res.text()
    expect(html).toContain('Événement complet')
    expect(html).toContain('lang="fr"')
  })

  it('renders French closed page', async () => {
    await seedEvent({ locale: 'fr', closes_at: '2020-01-01T00:00:00Z' })
    const res = await app.fetch(new Request('http://localhost/rsvp/test-i18n'), env)
    const html = await res.text()
    expect(html).toContain('Inscriptions terminées')
    expect(html).toContain('lang="fr"')
  })
